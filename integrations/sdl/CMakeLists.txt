cmake_minimum_required(VERSION 3.24)
project(MetroidvaniaStudioSdl VERSION 1.0.0 LANGUAGES C CXX)
option(STUDIO_BUILD_SAMPLE "Build the runnable room sample" ${PROJECT_IS_TOP_LEVEL})
option(STUDIO_FETCH_SDL "Download pinned SDL when a local SDL package is unavailable" ON)
find_package(SDL3 3.4 CONFIG QUIET)
if(NOT SDL3_FOUND)
    if(NOT STUDIO_FETCH_SDL)
        message(FATAL_ERROR "SDL 3.4+ is required. Set SDL3_DIR or enable STUDIO_FETCH_SDL.")
    endif()
    include(FetchContent)
    set(SDL_SHARED OFF CACHE BOOL "" FORCE)
    set(SDL_STATIC ON CACHE BOOL "" FORCE)
    set(SDL_TEST_LIBRARY OFF CACHE BOOL "" FORCE)
    set(SDL_TESTS OFF CACHE BOOL "" FORCE)
    set(SDL_EXAMPLES OFF CACHE BOOL "" FORCE)
    FetchContent_Declare(SDL3 URL https://www.libsdl.org/release/SDL3-3.4.16.tar.gz URL_HASH SHA256=7322236cd12090c3eb40b9728be4d49c76f66ad17d04369584d4ecad5cf77c68 DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
    FetchContent_MakeAvailable(SDL3)
endif()
add_library(metroidvania-studio-sdl STATIC src/StudioSdlRoom.cpp)
target_compile_features(metroidvania-studio-sdl PUBLIC cxx_std_17)
if(MSVC)
    target_compile_options(metroidvania-studio-sdl PUBLIC /utf-8)
endif()
target_include_directories(metroidvania-studio-sdl PUBLIC include)
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/../shared/native/StudioDocument.h")
    target_include_directories(metroidvania-studio-sdl PUBLIC ../shared/native)
endif()
target_link_libraries(metroidvania-studio-sdl PUBLIC SDL3::SDL3)
if(STUDIO_BUILD_SAMPLE)
    add_executable(metroidvania-studio-preview src-sample/main.cpp src-sample/sample-room.cpp)
    target_link_libraries(metroidvania-studio-preview PRIVATE metroidvania-studio-sdl)
    if(WIN32 AND TARGET SDL3::SDL3-shared)
        add_custom_command(TARGET metroidvania-studio-preview POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different $<TARGET_FILE:SDL3::SDL3-shared> $<TARGET_FILE_DIR:metroidvania-studio-preview>)
    endif()
endif()
